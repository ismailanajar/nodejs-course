4 · Live demo: three rooms, thirty lines (10 min)

npm install express
# check: "express": "^5.x" in package.json — v5 needs Node 18+, yours is 24 ✓

Create app.mjs — keep server.mjs, this one stands beside it:

// file: app.mjs
import express from "express";

const app = express();
app.use(express.json()); // bodies pre-parsed → req.body (chart 48, layer 1)

const tasks = [
  { id: "a1", title: "Learn routing", done: true },
  { id: "b2", title: "Learn bodies", done: false },
];

app.get("/health", (req, res) => res.json({ ok: true }));

app.get("/tasks", (req, res) => {
  const { done } = req.query;
  const list = done === undefined ? tasks : tasks.filter((t) => String(t.done) === done);
  res.json(list); // 200 + type + stringify, free
});

app.get("/tasks/:id", (req, res) => { // ← the pain remover: no manual split!
  const task = tasks.find((t) => t.id === req.params.id);
  if (!task) return res.status(404).json({ error: "unknown task" });
  res.json(task);
});

app.post("/tasks", (req, res) => {
  const title = req.body?.title; // already parsed — no chunks, no collector
  if (typeof title !== "string" || !title.trim())
    return res.status(400).json({ error: "title is required (non-empty string)" });
  const task = { id: `t${Date.now()}`, title: title.trim(), done: false };
  tasks.push(task);
  res.status(201).json(task);
});

app.listen(3000, () => console.log("express on http://localhost:3000"));
node app.mjs
node -e "console.log(await (await fetch('http://localhost:3000/tasks/a1')).json())"
# expected: { id: 'a1', title: 'Learn routing', done: true } — params, no splitting
node -e "const r = await fetch('http://localhost:3000/tasks',{method:'POST',headers:{'content-type':'application/json'},body:'{oops'}); console.log(r.status, await r.json())"
# expected: 400 + HTML?? — read §6 first box BEFORE running. Then run and compare.

Why it works: the router matched method+path (guard retired), :id captured the slot (splitter retired), express.json() parsed or rejected the parcel (collector retired). Your remaining code is pure decision.

6 · Common mistakes

“Corrupt JSON gives HTML, not my 400.” express.json() rejects bad parcels with its own HTML error before your handler runs. Fix (for now): notice it, don't fight it — lesson 0019 installs a JSON error translator; until then, valid-JSON clients only.
“Forgot app.use(express.json()) → req.body undefined.” Layers only dress what you install. Fix: the use line is load-bearing — no parser, no req.body, ever.
“Two replies: headers already sent.” res.json(a); res.json(b) without return — framework doesn't save you from fall-through. Fix: return res.status(...).json(...) on every early exit (lesson 0013's habit, unchanged).
